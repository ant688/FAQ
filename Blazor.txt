using Microsoft.AspNetCore.Components.Authorization;
using Microsoft.JSInterop;
using System.Security.Claims;

namespace BlazorAuthDemo.Services
{
    /// <summary>
    /// Demo provider: keeps the user in memory for the circuit and mirrors the username
    /// into sessionStorage so a browser refresh in the same tab stays logged in.
    /// </summary>
    public class CustomAuthStateProvider : AuthenticationStateProvider, IAsyncDisposable
    {
        private readonly IJSRuntime _js;
        private readonly Lazy<Task<IJSObjectReference>> _module;

        private ClaimsPrincipal _user = new(new ClaimsIdentity());

        private const string StorageKey = "BlazorAuthDemo:username";

        public CustomAuthStateProvider(IJSRuntime js)
        {
            _js = js;
            _module = new(() => _js.InvokeAsync<IJSObjectReference>(
                "import", "/authStorage.js").AsTask());
        }

        public override async Task<AuthenticationState> GetAuthenticationStateAsync()
        {
            if (_user.Identity?.IsAuthenticated == true)
                return new AuthenticationState(_user);

            var username = await GetStoredUsernameAsync();
            if (!string.IsNullOrWhiteSpace(username))
                SetUser(username);

            return new AuthenticationState(_user);
        }

        public async Task<bool> SignInAsync(string username, string password)
        {
            // DEMO ONLY:
            // Accept any non-empty username with password "123456"
            if (!string.IsNullOrWhiteSpace(username) && password == "123456")
            {
                SetUser(username);
                await StoreUsernameAsync(username);
                return true;
            }
            return false;
        }

        public async Task SignOutAsync()
        {
            _user = new ClaimsPrincipal(new ClaimsIdentity());
            await StoreUsernameAsync("");
            NotifyAuthenticationStateChanged(GetAuthenticationStateAsync());
        }

        private void SetUser(string username)
        {
            var identity = new ClaimsIdentity(new[]
            {
                new Claim(ClaimTypes.Name, username),
                new Claim(ClaimTypes.Role, "User")
            }, authenticationType: "DemoAuth");

            _user = new ClaimsPrincipal(identity);
            NotifyAuthenticationStateChanged(GetAuthenticationStateAsync());
        }

        private async Task StoreUsernameAsync(string username)
        {
            var mod = await _module.Value;
            await mod.InvokeVoidAsync("setSessionValue", StorageKey, username ?? "");
        }

        private async Task<string?> GetStoredUsernameAsync()
        {
            var mod = await _module.Value;
            return await mod.InvokeAsync<string?>("getSessionValue", StorageKey);
        }

        public async ValueTask DisposeAsync()
        {
            if (_module.IsValueCreated)
            {
                var mod = await _module.Value;
                await mod.DisposeAsync();
            }
        }
    }
}



export function setSessionValue(key, value) {
  sessionStorage.setItem(key, value ?? "");
}
export function getSessionValue(key) {
  return sessionStorage.getItem(key);
}
