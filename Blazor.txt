<!-- Components/AuthBootstrapper.razor -->
@inject Microsoft.JSInterop.IJSRuntime JS
@inject Authentication_example.Services.CustomAuthStateProvider Auth

@code {
    private bool _done;

    protected override async Task OnAfterRenderAsync(bool firstRender)
    {
        if (!firstRender || _done) return;

        var mod = await JS.InvokeAsync<Microsoft.JSInterop.IJSObjectReference>(
            "import", "/authStorage.js");

        var username = await mod.InvokeAsync<string>("getSessionValue", "BlazorAuthDemo:username");
        await Auth.RestoreAsync(username);

        await mod.DisposeAsync();
        _done = true;
        StateHasChanged(); // let UI update after auth restored
    }
}
